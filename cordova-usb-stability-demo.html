// UsbConnectionManager.kt — isolates native USB from Cordova bridge

class UsbConnectionManager(private val context: Context) {
    private val usbManager = context.getSystemService(Context.USB_SERVICE) as UsbManager
    private var connection: UsbDeviceConnection? = null
    private var usbInterface: UsbInterface? = null
    private var endpoint: UsbEndpoint? = null

    // Always close existing connection before reinitializing
    fun openDevice(device: UsbDevice): Boolean {
        closeDevice()
        if (!usbManager.hasPermission(device)) return false
        val conn = usbManager.openDevice(device) ?: return false
        val iface = device.getInterface(0)
        if (!conn.claimInterface(iface, true)) { conn.close(); return false }
        connection = conn; usbInterface = iface; endpoint = iface.getEndpoint(0)
        return true
    }

    // Returns -1 on failure — caller must trigger reconnect
    fun sendCommand(data: ByteArray): Int {
        val ep = endpoint ?: return -1
        val conn = connection ?: return -1
        return conn.bulkTransfer(ep, data, data.size, 5000)
    }

    // Safe teardown — always release interface before closing
    fun closeDevice() {
        try {
            usbInterface?.let { connection?.releaseInterface(it) }
            connection?.close()
        } catch (e: Exception) {
            Log.w("USB", "Close error: ${e.message}")
        } finally {
            connection = null; usbInterface = null; endpoint = null
        }
    }

    val isConnected get() = connection != null
}
// DebounceReconnectHandler.kt — handles bus resets & inrush disruptions

class DebounceReconnectHandler(
    private val connectionManager: UsbConnectionManager,
    private val debounceMs: Long = 500L,
    private val maxRetries: Int = 5
) {
    private val scope = CoroutineScope(Dispatchers.IO + SupervisorJob())
    private var reconnectJob: Job? = null
    private var retryCount = 0
    var onReconnected: (() -> Unit)? = null
    var onFailed: (() -> Unit)? = null

    // Called when bulkTransfer returns -1 or DETACH event fires
    fun onConnectionLost(device: UsbDevice) {
        reconnectJob?.cancel() // cancel any pending debounce — key to stability
        reconnectJob = scope.launch {
            delay(debounceMs)   // wait: collapses rapid bus resets into one event
            attemptReconnect(device)
        }
    }

    private suspend fun attemptReconnect(device: UsbDevice) {
        while (retryCount < maxRetries) {
            retryCount++
            if (connectionManager.openDevice(device)) {
                retryCount = 0
                withContext(Dispatchers.Main) { onReconnected?.invoke() }
                return
            }
            delay(200L * retryCount) // exponential backoff
        }
        withContext(Dispatchers.Main) { onFailed?.invoke() }
        retryCount = 0
    }

    fun cancel() { reconnectJob?.cancel(); scope.cancel() }
}
// UsbPermissionReceiver.kt — Android 13+ compliant permission receiver

class UsbPermissionReceiver: BroadcastReceiver() {

    companion object {
        const val ACTION = "com.plugin.usb.USB_PERMISSION"

        // Android 12+: FLAG_IMMUTABLE or FLAG_MUTABLE required
        fun buildPendingIntent(context: Context): PendingIntent {
            val intent = Intent(ACTION).apply { setPackage(context.packageName) }
            val flags = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S)
                PendingIntent.FLAG_IMMUTABLE else 0
            return PendingIntent.getBroadcast(context, 0, intent, flags)
        }
    }

    override fun onReceive(context: Context, intent: Intent) {
        if (ACTION != intent.action) return

        // API 33+: getParcelableExtra requires explicit class type
        val device: UsbDevice? = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU)
            intent.getParcelableExtra(UsbManager.EXTRA_DEVICE, UsbDevice::class.java)
        else
            @Suppress("DEPRECATION")
            intent.getParcelableExtra(UsbManager.EXTRA_DEVICE)

        val granted = intent.getBooleanExtra(UsbManager.EXTRA_PERMISSION_GRANTED, false)
        if (granted && device != null)
            UsbEventBus.post(UsbEvent.PermissionGranted(device))
        else
            UsbEventBus.post(UsbEvent.PermissionDenied)
    }
}
// www/usbPlugin.ts — TypeScript Cordova bridge layer

export interface UsbResult { success: boolean; message?: string; bytesWritten?: number }

const exec = (action: string, args: any[] = []) =>
    new Promise<UsbResult>((resolve, reject) =>
        (window as any).cordova.exec(resolve, (e: string) => reject(new Error(e)),
            'UsbManagerBridge', action, args))

export const UsbPlugin = {
    /** Initialize & open USB connection */
    connect: () => exec('connect'),

    /** Send raw bytes — base64 encoded for bridge */
    sendCommand: (data: Uint8Array) =>
        exec('sendCommand', [btoa(String.fromCharCode(...data))]),

    /** Manually trigger reconnect after UI-level error recovery */
    forceReconnect: () => exec('forceReconnect'),

    /** Disconnect and release all USB resources */
    disconnect: () => exec('disconnect'),
}

Debounce Reconnect Logic

// WITHOUT debounce — multiple failed reconnects:
reset #1
reset #2
reset #3
retry #1 fails
retry #2 fails
state corrupt
// WITH debounce (500ms window) — single stable reconnect:
reset #1
reset #2
reset #3
500ms debounce window
window closes
✅ one reconnect
✅ stable
✅ stable
Why this works: USB bus resets from inrush current often fire multiple DETACHED/ATTACHED events within 200–400ms. Each event without debouncing triggers a separate reconnect — racing against each other and corrupting the UsbDeviceConnection reference. The 500ms debounce window collapses all rapid events into one clean reinitialisation.

Android 13+ Compatibility Changes

BroadcastReceiver exported flag API 31+

Android 12+ requires all BroadcastReceivers to explicitly declare exported. Omitting it crashes the app at install.

<receiver android:name=".UsbPermissionReceiver">
<receiver android:name=".UsbPermissionReceiver"
android:exported="false">

getParcelableExtra deprecated API 33

The single-arg getParcelableExtra is deprecated in API 33. The type-safe overload must be used on newer devices.

intent.getParcelableExtra(EXTRA_DEVICE)
if (SDK_INT >= TIRAMISU)
intent.getParcelableExtra(EXTRA_DEVICE,
UsbDevice::class.java)
else @Suppress("DEPRECATION")
intent.getParcelableExtra(EXTRA_DEVICE)

PendingIntent flags required API 31+

PendingIntents must declare FLAG_IMMUTABLE or FLAG_MUTABLE on Android 12+. Missing this causes a runtime crash.

PendingIntent.getBroadcast(ctx, 0, intent, 0)
val flags = if (SDK_INT >= S)
PendingIntent.FLAG_IMMUTABLE else 0
PendingIntent.getBroadcast(ctx, 0, intent, flags)

USB permission persistence Best Practice

Permissions from the dialog do NOT persist across all restarts. Check hasPermission() every openDevice() call.

// Only check at startup — WRONG
if (firstLaunch) requestPermission(device)
 
// Check before every openDevice()
if (!usbManager.hasPermission(device)) {
requestPermission(device); return
}
